# Install everything in this file with:
#   pip install -r requirements.txt

# The web framework itself.
fastapi>=0.115.0

# The server that actually runs the app. The [standard] extra pulls in
# uvloop, httptools and websockets, which is what makes the --reload flag
# in the run command work properly.
uvicorn[standard]>=0.30.0

# The Supabase client. This talks to the project's REST API over HTTPS, so
# it is the only database dependency needed here.
#
# Chosen over SQLAlchemy on purpose. An ORM earns its keep when there are
# models, relationships and migrations to manage. This project has one
# table, created once by schema.sql, so an ORM plus a driver plus Alembic
# would be three more dependencies buying nothing.
supabase>=2.10.0

# Validates and shapes the request and response bodies in app/schemas.py.
# FastAPI depends on it already, but it is listed because our own code
# imports it directly, and a dependency you import is one you should name.
pydantic>=2.9.0

# Reads SUPABASE_URL, SUPABASE_KEY and JWT_SECRET out of the .env file so no
# secret is written in the source code.
python-dotenv>=1.0.0

# Signs and verifies the JWT access tokens issued at login. This is the
# library FastAPI's own security documentation uses, and the same one the
# BankingApp project settled on.
#
# Note that supabase already pulls PyJWT in as a transitive dependency, so
# this line installs nothing new today. It is listed anyway because our own
# code imports jwt directly, and a dependency you import is one you should
# name: otherwise a future version of supabase dropping it would break this
# app for reasons that look unrelated.
PyJWT>=2.9.0

# Hashes passwords before they are stored, and checks a typed password
# against a stored hash at login. The salt lives inside the hash.
#
# Used directly rather than through passlib. passlib adds a layer whose only
# real benefit is swapping algorithms later, its last release predates
# bcrypt 4, and the version mismatch between them is a well known source of
# a confusing warning at startup. One dependency less is the better trade
# here.
bcrypt>=4.2.0
